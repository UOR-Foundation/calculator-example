definition{
  name: calculate,
  parameters: [operation : Operation, left : Int64, right : Int64],
  result: (Result Int64 CalculatorError),
  body: (match operation
    | Operation.Add => (match (prim checked_add left right : (Option Int64))
      | Option.none => (ctor Result.error [Int64, CalculatorError] (ctor CalculatorError.Overflow))
      | Option.some value => (ctor Result.ok [Int64, CalculatorError] value))
    | Operation.Subtract => (match (prim checked_subtract left right : (Option Int64))
      | Option.none => (ctor Result.error [Int64, CalculatorError] (ctor CalculatorError.Overflow))
      | Option.some value => (ctor Result.ok [Int64, CalculatorError] value))
    | Operation.Multiply => (match (prim checked_multiply left right : (Option Int64))
      | Option.none => (ctor Result.error [Int64, CalculatorError] (ctor CalculatorError.Overflow))
      | Option.some value => (ctor Result.ok [Int64, CalculatorError] value))
    | Operation.Divide => (if (prim equal right (0 : Int64) : Bool) then (ctor Result.error [Int64, CalculatorError] (ctor CalculatorError.DivisionByZero)) else (match (prim checked_quotient left right : (Option Int64))
      | Option.none => (ctor Result.error [Int64, CalculatorError] (ctor CalculatorError.Overflow))
      | Option.some value => (ctor Result.ok [Int64, CalculatorError] value)))),
  axioms: []
}

definition{
  name: encodeCalculationResult,
  parameters: [value : (Result Int64 CalculatorError)],
  result: String,
  body: (match value
    | Result.error error => (match error
      | CalculatorError.DivisionByZero => "1\terror\tdivision-by-zero"
      | CalculatorError.Overflow => "1\terror\toverflow")
    | Result.ok result => (prim join (cons "1\tok\t" (cons (prim format_decimal result : String) (nil String))) "" : String)),
  axioms: [Classical.choice, Quot.sound, propext]
}

definition{
  name: dispatchString,
  parameters: [value : String],
  result: String,
  body: (match (decodeRequest value)
    | Result.error error => (encodeProtocolError error)
    | Result.ok request => (encodeCalculationResult (calculate (project request operation) (project request left) (project request right)))),
  axioms: [Classical.choice, Quot.sound, propext]
}

    | Option.some text => (prim utf8_encode (dispatchString text) : Bytes)),
  axioms: [Classical.choice, Quot.sound, propext]
}
